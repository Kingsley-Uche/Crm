<IfModule mod_rewrite.c>
    RewriteEngine On

    #
    # Serve files/directories from public/
    #
    RewriteCond %{REQUEST_URI} !^/public/

    RewriteCond %{DOCUMENT_ROOT}/dwello.netportal.com.ng/tap/public/$1 -f [OR]
    RewriteCond %{DOCUMENT_ROOT}/dwello.netportal.com.ng/tap/public/$1 -d
    RewriteRule ^(.*)$ public/$1 [L]

    #
    # Otherwise boot Laravel
    #
    RewriteRule ^(.*)$ public/index.php [L]
</IfModule>
